# Home 层次图（注释版）
# 入口文件：home/eurekaimer/home.nix
# 作用：这是 Home Manager 的总入口，只负责导入用户层模块。
#
# 上游入口：
# flake.nix
# └── home-manager.users.eurekaimer = import ./home/eurekaimer/home.nix
#
# unstable 规则：
# - nixpkgs-unstable 只在 flake.nix 中 import 一次，并命名为 pkgs-unstable。
# - Home Manager 模块通过 extraSpecialArgs 接收 pkgs-unstable。
# - 需要 unstable 软件包时，只在本模块参数中写 pkgs-unstable，并在下方标注用途。
# - 不要在 Home 模块中再次 import inputs.nixpkgs-unstable。

# 导入链说明：
# 1) modules/home/desktop.nix
#    含义：桌面会话层，管理窗口管理器、桌面壳、可选启动器/状态栏。
# 2) modules/home/core.nix
#    含义：核心使用层，管理 shell、终端、基础 UI、文件管理等高频基础组件。
# 3) modules/home/development.nix
#    含义：开发工具链层，按编辑器、通用 CLI、语言运行时、科研写作环境拆分管理。
# 4) modules/home/applications.nix
#    含义：应用层，集中日常软件；XDG 默认打开方式独立放在 mime-defaults.nix。

# /etc/nixos/home/eurekaimer/home.nix
# ├── ../../modules/home/desktop.nix
# │   ├── desktop/noctalia.nix         # Noctalia 壳层配置（顶部栏/控制中心等）
# │   │                                # 使用 inputs.noctalia.homeModules.default
# │   ├── desktop/niri.nix             # Niri 会话相关软件与配置
# │   ├── desktop/rofi.nix             # 可选启动器（默认未导入）
# │   └── desktop/waybar.nix           # 可选状态栏（默认未导入）
# ├── ../../modules/home/core.nix
# │   ├── core/shell.nix               # zsh/starship/zoxide 与 docker-ass 服务命令
# │   ├── core/kitty.nix               # kitty 终端与 LXGW 字体配置
# │   ├── core/fastfetch.nix           # fastfetch 信息展示
# │   ├── core/ui.nix                  # 图标主题/字体图标/GTK 光标主题
# │   └── core/yazi.nix                # yazi 文件管理器
# ├── ../../modules/home/development.nix
# │   ├── development/agents.nix       # AI coding agents（Codex/Claude Code）与 ACP 桥
# │   │                                # unstable: codex, codex-acp, claude-code, claude-code-acp
# │   ├── development/neovim.nix       # Neovim 与插件配置
# │   └── development/toolchain.nix    # 开发工具链总入口，按功能继续拆分
# │       ├── toolchain/editors.nix    # GUI 开发工具（VS Code/Zed/GitHub Desktop）与 Helix
# │       │                            # unstable: VS Code, Zed, Helix
# │       ├── toolchain/cli.nix        # 通用代码搜索/文件搜索工具（ripgrep/fd）
# │       ├── toolchain/shell.nix      # shell 语言服务器
# │       ├── toolchain/nix.nix        # Nix LSP 与格式化器
# │       ├── toolchain/lua.nix        # Lua LSP 与格式化器
# │       ├── toolchain/markdown.nix   # Markdown LSP
# │       ├── toolchain/python.nix     # uv/Jupyter/Pyright；Python 版本交给项目 .python-version
# │       ├── toolchain/javascript.nix # Node.js/pnpm
# │       ├── toolchain/go.nix         # Go/gopls/delve/gotools 与 GOPATH
# │       ├── toolchain/c-cpp.nix      # C/C++ 基础编译器
# │       └── toolchain/latex.nix      # TeX Live 精简中/英/日 + XeLaTeX + latexmk + texlab
# └── ../../modules/home/applications.nix
#     # 日常应用总入口；默认打开方式拆到 applications/mime-defaults.nix
#     ├── applications/knowledge.nix       # 知识管理（Obsidian/Zotero）
#     ├── applications/documents.nix       # 文档处理与阅读（PDF/EPUB/翻译）
#     ├── applications/media.nix           # 媒体播放与音乐
#     │                                    # unstable: mpv, go-musicfox
#     ├── applications/web.nix             # 浏览器与网络访问（Chrome/Clash Verge/Throne）
#     ├── applications/transfer.nix        # 下载/分发/图床上传

#     └── applications/mime-defaults.nix   # XDG 默认关联（PDF -> sioyek，图片 -> imv）
